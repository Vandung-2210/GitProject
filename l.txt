    <!-- Mã nguồn SCSS hoàn chỉnh tích hợp theo đúng các bước bài tập -->
    <script type="text/scss" id="scss-code">
        // ==========================================
        // BƯỚC 2: BẢNG BIẾN (abstracts/_variables.scss)
        // ==========================================
        $primary-color: red;
        $text-color: #222;
        $font-large: 32px;
        $font-medium: 28px;
        $padding-standard: 10px 20px;
        $box-shadow-default: 2px 2px 10px rgba(0, 0, 0, 0.1);

        // ==========================================
        // BƯỚC 3: MIXINS (abstracts/_mixins.scss)
        // ==========================================
        @mixin button-style($bg-color) {
            background: $bg-color;
            color: white;
            padding: $padding-standard;
            border-radius: 5px;
            border: none;
            cursor: pointer;
            font-weight: bold;
            transition: opacity 0.2s ease;

            &:hover {
                opacity: 0.85;
            }
        }

        @mixin box-style {
            background: white;
            padding: 15px;
            border-radius: 10px;
            box-shadow: $box-shadow-default;
        }

        // ==========================================
        // BƯỚC 4: ÁP DỤNG SCSS (base & components)
        // ==========================================
        * {
            margin: 0;
            padding: 0;
            box-sizing: border-box;
        }

        body {
            font-family: Arial, sans-serif;
            background-color: #f8f9fa;
            padding: 40px 20px;
        }

        .container {
            max-width: 800px;
            margin: 0 auto;
        }

        // base/_typography.scss
        h1 {
            font-size: $font-large;
            color: $text-color;
            margin-bottom: 10px;
        }

        h2 {
            font-size: $font-medium;
            color: $text-color;
            margin-bottom: 20px;
        }

        // components/_buttons.scss
        .button {
            @include button-style($primary-color);
        }

        // components/_cards.scss
        .card {
            @include box-style;
            margin-top: 20px;
        }
    </script>

    <!-- Script tự động biên dịch SCSS thành CSS khi load trang -->
    <script>
        const scssSource = document.getElementById('scss-code').text;
        Sass.compile(scssSource, function(result) {
            if (result.text) {
                document.getElementById('compiled-css').innerHTML = result.text;
            } else {
                console.error("Lỗi biên dịch SASS:", result);
            }
        });
    </script>
