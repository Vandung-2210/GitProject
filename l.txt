# 1. Tạo và di chuyển vào thư mục dự án
mkdir my-new-project
cd my-new-project

# 2. Khởi tạo Local Repository
git init

# 3. Liên kết với Remote Repository trên GitHub
git remote add origin https://github.com/dinhvandung/my-new-project.git

# 4. Tạo file README.md
echo "# My New Project" > README.md

# 5. Đưa file vào Staging Area và lưu Commit
git add README.md
git commit -m "Add README.md file"

# 6. Đổi tên nhánh mặc định thành main (nếu cần) và đẩy lên GitHub
git branch -M main
git push -u origin main
