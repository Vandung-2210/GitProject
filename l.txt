    <script>
        function changeWrap(value) {
            const container = document.getElementById('demoContainer');
            const statusText = document.getElementById('statusText');
            const buttons = document.querySelectorAll('.btn');

            // Cập nhật thuộc tính CSS
            container.style.flexWrap = value;

            // Đổi active button
            buttons.forEach(btn => btn.classList.remove('active'));
            event.target.classList.add('active');

            // Thông báo trạng thái
            const descriptions = {
                'nowrap': 'Đang áp dụng: flex-wrap: nowrap (Các item bị ép trên 1 hàng, không xuống dòng)',
                'wrap': 'Đang áp dụng: flex-wrap: wrap (Item 4, 5, 6 tự động xuống hàng tiếp theo)',
                'wrap-reverse': 'Đang áp dụng: flex-wrap: wrap-reverse (Hàng 2 nhảy lên trên, hàng 1 xuống dưới)',
                'initial': 'Đang áp dụng: flex-wrap: initial (Khôi phục giá trị mặc định nowrap)'
            };
            statusText.textContent = descriptions[value];
        }
    </script>
