# Bước 1 & 2: Kiểm tra môi trường Node.js & npm
node -v
# Output ví dụ: v18.14.0

npm -v
# Output ví dụ: 9.6.0

# Bước 3: Cài đặt SASS ở chế độ Global
npm install -g sass

# Bước 4: Kiểm tra cài đặt SASS thành công
sass -v
# Output ví dụ: 1.62.0

# Bước 5: Biên dịch 1 file duy nhất
sass scss/styles.scss css/styles.css

# Bước 6: Lệnh watch 1 file tự động
sass scss/styles.scss css/styles.css --watch

# Bước 7: Biên dịch & Watch toàn bộ thư mục SCSS sang CSS (Khuyên dùng)
sass scss:css --watch
