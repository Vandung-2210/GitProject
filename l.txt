    <!-- Mã nguồn SCSS hoàn chỉnh tích hợp theo chuẩn cấu trúc bài tập -->
    <script type="text/scss" id="scss-code">
        // ==========================================
        // 1. ABSTRACTS (_variables.scss & _mixins.scss)
        // ==========================================
        $primary-color: #007bff;
        $secondary-color: #6c757d;
        $text-color: #333333;
        $background-color: #f4f4f4;
        $border-color: #dddddd;
        $white-color: #ffffff;
        $box-shadow-val: 2px 2px 10px rgba(0, 0, 0, 0.1);

        @mixin button-style($bg-color) {
            background-color: $bg-color;
            color: $white-color;
            padding: 10px 20px;
            border: none;
            border-radius: 5px;
            cursor: pointer;
            font-weight: bold;
            transition: opacity 0.2s ease-in-out;

            &:hover {
                opacity: 0.85;
            }
        }

        @mixin box-shadow($shadow: $box-shadow-val) {
            box-shadow: $shadow;
        }

        // ==========================================
        // 2. BASE (_global.scss)
        // ==========================================
        * {
            margin: 0;
            padding: 0;
            box-sizing: border-box;
        }

        body {
            font-family: Arial, sans-serif;
            color: $text-color;
            background-color: $background-color;
            line-height: 1.6;
            padding: 50px 0;
        }

        // ==========================================
        // 3. LAYOUT (_container.scss)
        // ==========================================
        .container {
            width: 80%;
            max-width: 900px;
            margin: 0 auto;
        }

        // ==========================================
        // 4. COMPONENTS (_buttons.scss & _cards.scss)
        // ==========================================
        .button {
            @include button-style($primary-color);

            &--secondary {
                @include button-style($secondary-color);
            }
        }

        .card {
            background-color: $white-color;
            border: 1px solid $border-color;
            padding: 30px;
            border-radius: 8px;
            @include box-shadow;

            &__title {
                color: $primary-color;
                margin-bottom: 15px;
            }

            &__text {
                margin-bottom: 20px;
            }

            &__actions {
                display: flex;
                gap: 10px;
            }
        }
    </script>

    <!-- Script tự động biên dịch SCSS thành CSS khi load trang -->
    <script>
        const scssSource = document.getElementById('scss-code').text;
        Sass.compile(scssSource, function(result) {
            if (result.text) {
                document.getElementById('compiled-css').innerHTML = result.text;
            } else {
                console.error("Lỗi biên dịch SASS:", result);
            }
        });
    </script>
